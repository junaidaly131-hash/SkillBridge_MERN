# Everything worth indexing on SkillBridge is public and signed-out. The rest
# of the app is behind authentication, so a crawler reaching it gets a login
# redirect at best and a useless duplicate of it at worst - those paths are
# listed below so the crawl budget goes to pages that can actually rank.
#
# This file lives in public/ so Vite copies it to the site root verbatim. Put it
# anywhere else and Vercel's SPA rewrite answers /robots.txt with index.html,
# and crawlers read an HTML page as a rules file - which they ignore entirely.

User-agent: *

# Signed-in application
Disallow: /dashboard
Disallow: /profile
Disallow: /chat
Disallow: /feedback
Disallow: /support
Disallow: /credits
Disallow: /meetings
Disallow: /admin

# Account flows - nothing to rank, and /verify carries one-time codes
Disallow: /verify
Disallow: /forgot-password

Allow: /

Sitemap: https://www.skill-bridge.me/sitemap.xml
